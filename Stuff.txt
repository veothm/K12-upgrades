  let currentWidgetTimeZone = null;

  function getWeatherCondition(code) {
    if (code === 0) return 'CLEAR';
    if (code >= 1 && code <= 3) return 'CLOUDY';
    if (code >= 51 && code <= 67) return 'RAIN';
    if (code >= 71 && code <= 77) return 'SNOW';
    if (code >= 80 && code <= 82) return 'RAIN';
    if (code >= 95) return 'THUNDER';
    return 'RAIN';
  }

  function startDynamicClock() {
    setInterval(() => {
      const weatherWidgets = document.querySelectorAll('[data-slot-index]');
      weatherWidgets.forEach((widgetSlot) => {
        const timeDigitElement = widgetSlot.querySelector('.font-bold.tracking-tighter');
        const ampmElement = widgetSlot.querySelector('.font-semibold.text-2xl');
        if (!timeDigitElement) return;

        const now = new Date();
        const options = { hour: 'numeric', minute: '2-digit', second: '2-digit', hour12: true };
        if (currentWidgetTimeZone) options.timeZone = currentWidgetTimeZone;

        const localTimeStr = now.toLocaleTimeString('en-US', options);
        const [timePart, ampm] = localTimeStr.split(' ');
        const timeWithoutSeconds = timePart.split(':').slice(0, 2).join(':');

        timeDigitElement.textContent = timeWithoutSeconds;
        if (ampmElement && ampm) ampmElement.textContent = ampm;
      });
    }, 1000);
  }

  async function updateWidgetByZip(widgetSlot, zipCode) {
    try {
      const geoRes = await fetch(`https://api.zippopotam.us/us/${zipCode}`);
      if (!geoRes.ok) throw new Error('Invalid ZIP Code');
      const geoData = await geoRes.json();

      const place = geoData.places[0];
      const cityName = place['place name'];
      const lat = place.latitude;
      const lon = place.longitude;

      // Open-Meteo expects either fahrenheit or celsius
      const apiUnit = activeTempUnit === 'fahrenheit' ? 'fahrenheit' : 'celsius';

      const weatherRes = await fetch(
        `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,weather_code&temperature_unit=${apiUnit}&timezone=auto`
      );
      const weatherData = await weatherRes.json();

      let rawTemp = weatherData.current.temperature_2m;
      let unitSymbol = '°';

      if (activeTempUnit === 'kelvin') {
        rawTemp = rawTemp + 273.15; // Convert Celsius to Kelvin
        unitSymbol = ' K';
      }

      const temp = Math.round(rawTemp * 10) / 10;
      const weatherCode = weatherData.current.weather_code;
      const condition = getWeatherCondition(weatherCode);

      currentWidgetTimeZone = weatherData.timezone;

      const locationSpan = widgetSlot.querySelector('.lucide-map-pin + span') || widgetSlot.querySelector('svg.lucide-map-pin ~ span');
      const tempElement = widgetSlot.querySelector('.font-bold.\\@lg\\:text-\\[48px\\]') || widgetSlot.querySelector('.font-bold.text-3xl');
      const conditionElement = widgetSlot.querySelector('p.font-semibold') || widgetSlot.querySelector('.uppercase.font-semibold');

      if (locationSpan) {
        locationSpan.textContent = cityName;
        widgetSlot.setAttribute('data-applied-zip', `${zipCode}-${activeTempUnit}`);
      }
      if (tempElement) tempElement.textContent = `${temp}${unitSymbol}`;
      if (conditionElement) conditionElement.textContent = condition;
    } catch (err) {
      console.error('Could not update weather:', err);
    }
  }

  function updateAllWeatherWidgets(zipCode) {
    const slots = document.querySelectorAll('[data-slot-index]');
    slots.forEach((slot) => {
      const isWeather = slot.querySelector('div[class*="weather-widget"]') || slot.querySelector('.lucide-map-pin');
      if (isWeather && slot.getAttribute('data-applied-zip') !== `${zipCode}-${activeTempUnit}`) {
        updateWidgetByZip(slot, zipCode);
      }
    });
  }

  function renderPalettes(filterText = "") {
    const container = document.getElementById('k12-palette-list');
    if (!container) return;

    const filtered = PRESET_PALETTES.filter(p => p.name.toLowerCase().includes(filterText.toLowerCase()));

    if (filtered.length === 0) {
      container.innerHTML = `<p class="text-xs text-neutral-400 py-4 text-center col-span-2">No palettes match "${filterText}"</p>`;
      return;
    }

    container.innerHTML = filtered.map(p => `
      <div class="p-3 bg-white/5 border border-white/10 rounded-xl flex items-center justify-between hover:border-white/30 transition-all">
        <div>
          <p class="text-xs font-bold text-white mb-1.5">${p.name}</p>
          <div class="flex items-center gap-1">
            <span class="w-3.5 h-3.5 rounded-full border border-white/20 inline-block" style="background:${p.cardBg}" title="Card Base"></span>
            <span class="w-3.5 h-3.5 rounded-full border border-white/20 inline-block" style="background:${p.activeBg}" title="Active Tab"></span>
            <span class="w-3.5 h-3.5 rounded-full border border-white/20 inline-block" style="background:${p.inactiveBg}" title="Inactive Tab"></span>
          </div>
        </div>
        <button type="button" data-palette-name="${p.name}" class="k12-apply-btn text-xs font-semibold px-2.5 py-1 bg-blue-600/80 hover:bg-blue-500 rounded-lg text-white transition-colors cursor-pointer">
          Apply
        </button>
      </div>
    `).join('');

      btn.addEventListener('click', () => {
        const pName = btn.getAttribute('data-palette-name');
        const palette = PRESET_PALETTES.find(item => item.name === pName);
        if (!palette) return;

        document.getElementById('k12-card-bg-picker').value = palette.cardBg;
        document.getElementById('k12-card-bg-hex').value = palette.cardBg;
        document.getElementById('k12-card-text-picker').value = palette.cardText;
        document.getElementById('k12-card-text-hex').value = palette.cardText;

        document.getElementById('k12-active-tab-bg-picker').value = palette.activeBg;
        document.getElementById('k12-active-tab-bg-hex').value = palette.activeBg;
        document.getElementById('k12-active-tab-text-picker').value = palette.activeText;
        document.getElementById('k12-active-tab-text-hex').value = palette.activeText;

        document.getElementById('k12-inactive-tab-bg-picker').value = palette.inactiveBg;
        document.getElementById('k12-inactive-tab-bg-hex').value = palette.inactiveBg;
        document.getElementById('k12-inactive-tab-text-picker').value = palette.inactiveText;
        document.getElementById('k12-inactive-tab-text-hex').value = palette.inactiveText;
      });
    });
  }

  function setupSectionTabs() {
    const tabs = document.querySelectorAll('.k12-nav-tab');
    const sections = document.querySelectorAll('.k12-tab-section');

    tabs.forEach(tab => {
      tab.addEventListener('click', () => {
        tabs.forEach(t => t.classList.remove('active'));
        sections.forEach(s => s.classList.add('hidden'));

        tab.classList.add('active');
        const targetId = tab.getAttribute('data-target');
        const targetSection = document.getElementById(targetId);
        if (targetSection) targetSection.classList.remove('hidden');
      });
    });
  }

  function exportConfig() {
    const config = {
      zip: localStorage.getItem('k12_saved_zip') || '10001',
      tempUnit: localStorage.getItem('k12_temp_unit') || 'fahrenheit',
      font: localStorage.getItem('k12_saved_font') || '',
      cardBg: localStorage.getItem('k12_card_bg_color') || '#262626',
      cardText: localStorage.getItem('k12_card_text_color') || '#ffffff',
      activeTabBg: localStorage.getItem('k12_active_tab_bg') || '#333333',
      activeTabText: localStorage.getItem('k12_active_tab_text') || '#ffffff',
      inactiveTabBg: localStorage.getItem('k12_inactive_tab_bg') || '#1d1d1d',
      inactiveTabText: localStorage.getItem('k12_inactive_tab_text') || '#a3a3a3'
    };

    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(config, null, 2));
    const downloadAnchor = document.createElement('a');
    downloadAnchor.setAttribute("href", dataStr);
    downloadAnchor.setAttribute("download", `k12_theme_config_${new Date().toISOString().slice(0, 10)}.json`);
    document.body.appendChild(downloadAnchor);
    downloadAnchor.click();
    downloadAnchor.remove();
  }

  function importConfig(jsonString) {
    try {
      const config = JSON.parse(jsonString);
      if (config.zip) localStorage.setItem('k12_saved_zip', config.zip);
      if (config.tempUnit) localStorage.setItem('k12_temp_unit', config.tempUnit);
      if (config.font) localStorage.setItem('k12_saved_font', config.font);
      if (config.cardBg) localStorage.setItem('k12_card_bg_color', config.cardBg);
      if (config.cardText) localStorage.setItem('k12_card_text_color', config.cardText);
      if (config.activeTabBg) localStorage.setItem('k12_active_tab_bg', config.activeTabBg);
      if (config.activeTabText) localStorage.setItem('k12_active_tab_text', config.activeTabText);
      if (config.inactiveTabBg) localStorage.setItem('k12_inactive_tab_bg', config.inactiveTabBg);
      if (config.inactiveTabText) localStorage.setItem('k12_inactive_tab_text', config.inactiveTabText);

      location.reload();
    } catch (e) {
      alert("Invalid configuration file format.");
    }
  }
